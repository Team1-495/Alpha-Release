name: TASP Delivery Platform Assembly Line

on:
  pull_request:
    branches: [ develop, main ]

jobs:
  quality-gate:
    runs-on: ubuntu-latest
    steps:
      - name: Fetch Current Branch Assets
        uses: actions/checkout@v4

      - name: Bootstrap Python Infrastructure
        uses: actions/setup-python@v5
        with:
          python-version: '3.11'

      - name: Build System Dependencies
        run: |
          pip install fastapi uvicorn pydantic pytest numpy httpx

      - name: Run Algorithmic Testing Checks
        run: |
          pytest -v backend/tests/
